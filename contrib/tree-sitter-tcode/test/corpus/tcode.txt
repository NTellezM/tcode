fn area(f: Forma) -> f64 {
    match f {
        Forma.Circulo(r) -> { return 3.14 * r * r; }
        Forma.Rectangulo(a, b) | Forma.Nada -> { return a * b; }
        _ -> { return 0.0; }
    }
}
